<script type="text/javascript">
	$(document).ready(function() {
	  var owl = $('#activity');
	  owl.owlCarousel({
		items: 2,
		loop: true,
		nav:true,
			dots:true,
		margin: 10,
		autoplay: true,
		autoplayTimeout: 5000,
		autoplayHoverPause: true,
		responsive:{
			0:{
			  items:1
			},
			600:{
			  items:2
			},
			1000:{
			  items:2
			}
		}
	  });
	  $('#activity_play').on('click', function() {
		owl.trigger('play.owl.autoplay', [5000])
	  })
	  $('#activity_stop').on('click', function() {
		owl.trigger('stop.owl.autoplay')
	  })
	})
</script>
